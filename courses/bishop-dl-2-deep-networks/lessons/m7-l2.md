## 1. More arithmetic (inside the class)

Subtraction, division and the "reflected" versions, so that `1 - x` and `2 / x` work too. Division is built from
multiplication and powers, so it needs no backward function of its own.

<!-- skip -->
```python
    def __neg__(self):
        return self * -1.0

    def __sub__(self, other):
        return self + (-Tensor.lift(other))

    def __rsub__(self, other):
        return Tensor.lift(other) - self

    def __truediv__(self, other):
        return self * Tensor.lift(other) ** -1.0

    def __rtruediv__(self, other):
        return Tensor.lift(other) * self ** -1.0

    __radd__ = __add__
    __rmul__ = __mul__
```

## 2. A matrix product for batches (inside the class, replacing the old one)

The same rule as before, `ā = ḡ bᵀ` and `b̄ = aᵀ ḡ`, written so it also works for stacks of matrices (you'll need that for
attention in part 3).

<!-- skip -->
```python
    def __matmul__(self, other):
        other = Tensor.lift(other)
        def back(g):
            a, b = self.data, other.data
            ga = g @ np.swapaxes(b, -1, -2)
            gb = np.swapaxes(a, -1, -2) @ g
            self._add_grad(_unbroadcast(ga, a.shape)); other._add_grad(_unbroadcast(gb, b.shape))
        return self._make(self.data @ other.data, (self, other), back)
```

## 3. Elementwise functions (inside the class)

Each one stores what its backward pass needs. Notice how `exp` and `sigmoid` reuse their own output: the derivative of
eˣ is eˣ, and the derivative of σ(a) is σ(1 − σ).

<!-- skip -->
```python
    def exp(self):
        out = np.exp(self.data)
        return self._make(out, (self,), lambda g: self._add_grad(g * out))

    def log(self):
        return self._make(np.log(self.data), (self,), lambda g: self._add_grad(g / self.data))

    def tanh(self):
        out = np.tanh(self.data)
        return self._make(out, (self,), lambda g: self._add_grad(g * (1 - out ** 2)))

    def relu(self):
        mask = self.data > 0
        return self._make(self.data * mask, (self,), lambda g: self._add_grad(g * mask))

    def sigmoid(self):
        out = 1 / (1 + np.exp(-self.data))
        return self._make(out, (self,), lambda g: self._add_grad(g * out * (1 - out)))
```

## 4. Reductions, shapes and indexing (inside the class, replacing the old `sum`)

- `sum` along an axis sends the gradient back to every element that was added: a **broadcast**.
- `max` sends the gradient only to the element(s) that won.
- `reshape` and `transpose` just rearrange the gradient back.
- Indexing (`x[rows, cols]`) sends each gradient to the place it came from. `np.add.at` adds correctly even when the
  same element is picked twice.

<!-- skip -->
```python
    def sum(self, axis=None, keepdims=False):
        def back(g):
            if axis is not None and not keepdims:
                g = np.expand_dims(g, axis)
            self._add_grad(np.broadcast_to(g, self.shape).copy())
        return self._make(self.data.sum(axis=axis, keepdims=keepdims), (self,), back)

    def mean(self, axis=None, keepdims=False):
        n = self.data.size if axis is None else self.data.shape[axis]
        return self.sum(axis, keepdims) * (1.0 / n)

    def max(self, axis=-1, keepdims=True):
        out = self.data.max(axis=axis, keepdims=True)
        mask = (self.data == out)
        mask = mask / mask.sum(axis=axis, keepdims=True)
        def back(g):
            if not keepdims:
                g = np.expand_dims(g, axis)
            self._add_grad(g * mask)
        return self._make(out if keepdims else out.squeeze(axis), (self,), back)

    def reshape(self, *shape):
        return self._make(self.data.reshape(*shape), (self,), lambda g: self._add_grad(g.reshape(self.shape)))

    def transpose(self, *axes):
        axes = axes or tuple(range(self.data.ndim))[::-1]
        inv = np.argsort(axes)
        return self._make(self.data.transpose(axes), (self,), lambda g: self._add_grad(g.transpose(inv)))

    def __getitem__(self, idx):
        def back(g):
            full = np.zeros_like(self.data)
            np.add.at(full, idx, g)
            self._add_grad(full)
        return self._make(self.data[idx], (self,), back)
```

## 5. Functions built from the pieces (after the class)

`log_softmax` is the careful way to compute the log of a softmax (section 5.4.4): subtracting the largest logit first
changes nothing mathematically but avoids overflow in `exp`. `cross_entropy` picks out each example's log-probability of
its true class.

<!-- skip -->
```python
def concat(tensors, axis=-1):
    data = np.concatenate([t.data for t in tensors], axis=axis)
    sizes = np.cumsum([t.shape[axis] for t in tensors])[:-1]
    def back(g):
        for t, part in zip(tensors, np.split(g, sizes, axis=axis)):
            t._add_grad(part)
    return tensors[0]._make(data, tuple(tensors), back)


def log_softmax(x: Tensor, axis=-1) -> Tensor:
    shifted = x - Tensor(x.data.max(axis=axis, keepdims=True))     # a constant shift, for numerical safety
    return shifted - shifted.exp().sum(axis=axis, keepdims=True).log()


def softmax(x: Tensor, axis=-1) -> Tensor:
    return log_softmax(x, axis).exp()


def cross_entropy(logits: Tensor, labels) -> Tensor:
    """Mean negative log-probability of the right class (Bishop 5.4.4, 6.4.3). labels: integer array."""
    lp = log_softmax(logits, axis=-1)
    labels = np.asarray(labels)
    return -lp[np.arange(len(labels)), labels].mean()


def mse(pred: Tensor, target) -> Tensor:
    return ((pred - Tensor.lift(target)) ** 2).mean()
```

## 6. Layers (after the class)

`Module.parameters()` finds every trainable Tensor inside a model, including those in nested layers and lists of layers.
`Linear` uses **He initialization** (section 7.2.5): Gaussian weights with variance 2 / (number of inputs).

<!-- skip -->
```python
class Module:
    def parameters(self):
        out = []
        for v in vars(self).values():
            if isinstance(v, Tensor) and v.requires_grad:
                out.append(v)
            elif isinstance(v, Module):
                out += v.parameters()
            elif isinstance(v, (list, tuple)):
                for item in v:
                    if isinstance(item, Module):
                        out += item.parameters()
        return out

    def __call__(self, *args):
        return self.forward(*args)


class Linear(Module):
    def __init__(self, n_in, n_out, rng, bias=True):
        self.W = Tensor(rng.normal(0, np.sqrt(2.0 / n_in), (n_in, n_out)), requires_grad=True)   # He initialization
        self.b = Tensor(np.zeros(n_out), requires_grad=True) if bias else None

    def forward(self, x):
        y = x @ self.W
        return y + self.b if self.b is not None else y


class MLP(Module):
    def __init__(self, sizes, rng):
        self.layers = [Linear(a, b, rng) for a, b in zip(sizes[:-1], sizes[1:])]

    def forward(self, x):
        for layer in self.layers[:-1]:
            x = layer(x).relu()
        return self.layers[-1](x)
```

## 7. Optimizers (after the class)

SGD with momentum and Adam, exactly as in the algorithm boxes of section 7.3. The Adam version uses **decoupled**
weight decay (shrinking the weights directly, as in the AdamW method) instead of adding λw to the gradient.

<!-- skip -->
```python
class SGD:
    def __init__(self, params, lr=0.1, momentum=0.0, weight_decay=0.0):
        self.params, self.lr, self.mu, self.wd = params, lr, momentum, weight_decay
        self.v = [np.zeros_like(p.data) for p in params]

    def step(self):
        for p, v in zip(self.params, self.v):
            g = p.grad + self.wd * p.data
            v *= self.mu
            v -= self.lr * g
            p.data += v

    def zero_grad(self):
        for p in self.params:
            p.grad = None


class Adam:
    def __init__(self, params, lr=1e-3, betas=(0.9, 0.999), eps=1e-8, weight_decay=0.0):
        self.params, self.lr, (self.b1, self.b2), self.eps, self.wd = params, lr, betas, eps, weight_decay
        self.m = [np.zeros_like(p.data) for p in params]
        self.s = [np.zeros_like(p.data) for p in params]
        self.t = 0

    def step(self):
        self.t += 1
        for p, m, s in zip(self.params, self.m, self.s):
            g = p.grad
            m *= self.b1; m += (1 - self.b1) * g
            s *= self.b2; s += (1 - self.b2) * g * g
            mhat = m / (1 - self.b1 ** self.t)
            shat = s / (1 - self.b2 ** self.t)
            p.data -= self.lr * (mhat / (np.sqrt(shat) + self.eps) + self.wd * p.data)   # decoupled weight decay

    def zero_grad(self):
        for p in self.params:
            p.grad = None
```

## 8. Check every gradient

A gradient check that touches every new operation at once: matrix product, tanh, sigmoid, concat, log-softmax and
cross-entropy, powers, mean, max, indexing and exp.

```python
import numpy as np
from minigrad import Tensor, concat, cross_entropy

rng = np.random.default_rng(0)
A = rng.normal(size=(3, 4))
B = rng.normal(size=(4, 2))
labels = np.array([0, 1, 1])

def f(a, track=False):
    x = Tensor(a, requires_grad=track)
    h = (x @ Tensor(B)).tanh()
    loss = (cross_entropy(concat([h, h.sigmoid() * 2], axis=-1), labels)
            + (x ** 2).mean() + x.max(axis=1).sum() + x[:, 1:3].exp().sum())
    return x, loss

x, loss = f(A, track=True)
loss.backward()

numeric = np.zeros_like(A)
for i in np.ndindex(A.shape):
    up, down = A.copy(), A.copy()
    up[i] += 1e-6; down[i] -= 1e-6
    numeric[i] = (float(f(up)[1].data) - float(f(down)[1].data)) / 2e-6
print(f"largest gradient error: {np.abs(x.grad - numeric).max():.2e}")
```

```output
largest gradient error: 1.03e-09
```

If you see something around 10⁻⁹ or smaller, every operation's backward pass agrees with numerical differentiation.
