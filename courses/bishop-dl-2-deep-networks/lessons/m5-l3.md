# Build: minigrad, part 1: the Tensor and its backward pass

Today you write the heart of your library: a `Tensor` that remembers how it was computed and can send gradients back.
Put this code in a file called `minigrad.py` (or a notebook cell); you'll extend it on day 7.

## 1. Forward mode first, in five lines

Before reverse mode, see forward mode from section 8.2.1 with **dual numbers**: each value carries its derivative along.

```python
import numpy as np

class Dual:
    def __init__(self, val, der): self.val, self.der = val, der
    def __add__(self, o): return Dual(self.val + o.val, self.der + o.der)
    def __mul__(self, o): return Dual(self.val * o.val, self.der * o.val + self.val * o.der)
    def sin(self): return Dual(np.sin(self.val), np.cos(self.val) * self.der)

x1, x2 = Dual(2.0, 1.0), Dual(3.0, 0.0)          # seed: derivative with respect to x1
f = x1 * x2 + x1.sin()                           # f = x1·x2 + sin(x1)
print("f =", round(f.val, 4), "  df/dx1 =", round(f.der, 4), "  (exact: x2 + cos(x1) =", round(3 + np.cos(2), 4), ")")
```

```output
f = 6.9093   df/dx1 = 2.5839   (exact: x2 + cos(x1) = 2.5839 )
```

One sweep, one input's derivative. To get df/dx2 you'd have to run again with the seed on x2. Reverse mode gets both at
once.

## 2. The Tensor

Each Tensor stores its data, its gradient (the adjoint), its parents in the graph, and a small function `_back` that
knows how to send gradients to those parents.

```python
def _unbroadcast(grad, shape):
    """NumPy broadcasting copies a value along some axes; its gradient must be summed back over them."""
    while grad.ndim > len(shape):
        grad = grad.sum(axis=0)
    for i, n in enumerate(shape):
        if n == 1 and grad.shape[i] != 1:
            grad = grad.sum(axis=i, keepdims=True)
    return grad


class Tensor:
    def __init__(self, data, requires_grad=False, _parents=(), _back=None):
        self.data = np.asarray(data, dtype=np.float64)
        self.requires_grad = requires_grad
        self.grad = None
        self._parents = _parents
        self._back = _back

    shape = property(lambda self: self.data.shape)

    def _make(self, data, parents, back):
        need = any(p.requires_grad for p in parents)
        return Tensor(data, need, parents if need else (), back if need else None)

    @staticmethod
    def lift(x):
        return x if isinstance(x, Tensor) else Tensor(x)

    def _add_grad(self, g):                       # adjoints are ADDED (a value can be used many times)
        if self.requires_grad:
            self.grad = g if self.grad is None else self.grad + g

    def __add__(self, other):
        other = Tensor.lift(other)
        def back(g):
            self._add_grad(_unbroadcast(g, self.shape)); other._add_grad(_unbroadcast(g, other.shape))
        return self._make(self.data + other.data, (self, other), back)

    def __mul__(self, other):
        other = Tensor.lift(other)
        def back(g):
            self._add_grad(_unbroadcast(g * other.data, self.shape))
            other._add_grad(_unbroadcast(g * self.data, other.shape))
        return self._make(self.data * other.data, (self, other), back)

    def __pow__(self, k):
        return self._make(self.data ** k, (self,), lambda g: self._add_grad(g * k * self.data ** (k - 1)))

    def __matmul__(self, other):
        other = Tensor.lift(other)
        def back(g):
            self._add_grad(g @ other.data.T); other._add_grad(self.data.T @ g)
        return self._make(self.data @ other.data, (self, other), back)

    def __neg__(self):
        return self * -1.0

    def __sub__(self, other):
        return self + (-Tensor.lift(other))

    __radd__ = __add__
    __rmul__ = __mul__

    def tanh(self):
        out = np.tanh(self.data)
        return self._make(out, (self,), lambda g: self._add_grad(g * (1 - out ** 2)))

    def relu(self):
        mask = self.data > 0
        return self._make(self.data * mask, (self,), lambda g: self._add_grad(g * mask))

    def sum(self):
        return self._make(self.data.sum(), (self,), lambda g: self._add_grad(np.ones_like(self.data) * g))

    def backward(self):
        order, seen = [], set()
        def visit(t):                              # topological order: parents before children
            if id(t) not in seen:
                seen.add(id(t))
                for p in t._parents:
                    visit(p)
                order.append(t)
        visit(self)
        self.grad = np.ones_like(self.data)        # d(output)/d(output) = 1
        for t in reversed(order):                  # reverse sweep (section 8.2.2)
            if t._back is not None and t.grad is not None:
                t._back(t.grad)

print("Tensor is ready")
```

```output
Tensor is ready
```

Read each `back` function and match it to an adjoint rule from the reading. The only new piece is `_unbroadcast`: when
NumPy adds a bias vector to every row of a matrix, the bias was effectively copied, so its gradient is the **sum** over
the rows.

## 3. Check it on the book's kind of example

`f = x1·x2 + tanh(x1)`. Reverse mode gives both derivatives in one backward sweep.

```python
x1 = Tensor(2.0, requires_grad=True)
x2 = Tensor(3.0, requires_grad=True)
f = x1 * x2 + x1.tanh()
f.backward()
print("df/dx1 =", round(float(x1.grad), 6), "  exact:", round(3 + (1 - np.tanh(2) ** 2), 6))
print("df/dx2 =", round(float(x2.grad), 6), "  exact:", 2.0)
```

```output
df/dx1 = 3.070651   exact: 3.070651
df/dx2 = 2.0   exact: 2.0
```

## 4. Check it on yesterday's network

Rebuild the two-layer tanh network with Tensors and compare every gradient with numerical differentiation.

```python
rng = np.random.default_rng(0)
X = rng.random((20, 1)); T = np.sin(2 * np.pi * X)
init = {"W1": rng.normal(0, 1, (1, 8)), "b1": np.zeros(8), "W2": rng.normal(0, 0.35, (8, 1)), "b2": np.zeros(1)}

def loss_fn(p):
    hidden = (Tensor(X) @ p["W1"] + p["b1"]).tanh()
    y = hidden @ p["W2"] + p["b2"]
    return ((y - T) ** 2).sum() * 0.5

params = {k: Tensor(v.copy(), requires_grad=True) for k, v in init.items()}
loss_fn(params).backward()

def numeric_grad(name, idx, eps=1e-6):
    out = []
    for s in (+eps, -eps):
        p = {k: Tensor(v.copy()) for k, v in init.items()}
        p[name].data[idx] += s
        out.append(float(loss_fn(p).data))
    return (out[0] - out[1]) / (2 * eps)

worst = max(abs(numeric_grad(n, i) - params[n].grad[i]) for n in init for i in np.ndindex(init[n].shape))
print(f"largest difference from numerical gradients: {worst:.2e}")
```

```output
largest difference from numerical gradients: 1.34e-09
```

Your library now computes exact gradients for **any** network you build from these operations, with no calculus by
hand. Notice that `loss_fn` contains no derivative code at all.

## 5. Train with it

```python
lr = 0.005
X = rng.random((50, 1)); T = np.sin(2 * np.pi * X) + rng.normal(0, 0.1, (50, 1))
for step in range(4001):
    for p in params.values():
        p.grad = None                              # gradients add up, so clear them before each backward pass
    loss = loss_fn(params)
    loss.backward()
    for p in params.values():
        p.data -= lr * p.grad
    if step % 1000 == 0:
        print(f"step {step:4d}: error {float(loss.data):.3f}")
```

```output
step    0: error 12.811
step 1000: error 3.844
step 2000: error 1.612
step 3000: error 0.304
step 4000: error 0.303
```

Forgetting to clear the gradients is the classic automatic-differentiation bug: they'd keep piling up from every step.
That's why every framework has a `zero_grad()`. On day 7 you'll add layers, a softmax cross-entropy and Adam, and train on
real images.
