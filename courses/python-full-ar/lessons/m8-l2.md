# المجموعات وre.sub وتنظيف البيانات

تصنع الأقواس **مجموعة**: جزءًا من المطابقة تستطيع استخراجه. وتعطي `re.findall` كل المطابقات، وتستبدلها `re.sub`:

```python
import re

time = "14:05"
if matches := re.search(r"^(\d{1,2}):(\d{2})$", time):   # := stores and checks at once
    hours, minutes = int(matches.group(1)), int(matches.group(2))
    print(hours * 60 + minutes, "minutes after midnight")    # 845

url = "https://www.github.com/mona-dev"
print(re.sub(r"^(https?://)?(www\.)?github\.com/", "", url))   # mona-dev

text = "Meet on 2026-11-14 or 2026-12-01"
print(re.findall(r"\d{4}-\d{2}-\d{2}", text))   # ['2026-11-14', '2026-12-01']
```
