```python
# triagebot/api.py
"""Tickets in, decisions out. Nothing reaches a customer until a person approves it."""
import json
import sqlite3
from contextlib import asynccontextmanager

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

from triagebot import tools
from triagebot.agent import triage
from triagebot.llm import chat
from triagebot.search import Searcher          # search.py copied from DocAnswer

QUEUE_PATH = "data/queue.db"


@asynccontextmanager
async def lifespan(app):
    tools.HELP_CENTER = Searcher("data/help_chunks.jsonl")
    with sqlite3.connect(QUEUE_PATH) as db:
        db.execute("CREATE TABLE IF NOT EXISTS queue (id TEXT PRIMARY KEY, decision TEXT, status TEXT)")
    yield


app = FastAPI(title="TriageBot", lifespan=lifespan)


class Ticket(BaseModel):
    id: str
    email: str
    text: str


@app.post("/tickets")
def new_ticket(ticket: Ticket):
    decision = triage(ticket.model_dump(), chat, trace_path="data/traces.jsonl")
    with sqlite3.connect(QUEUE_PATH) as db:
        db.execute("INSERT OR REPLACE INTO queue VALUES (?, ?, 'pending')", (ticket.id, json.dumps(decision)))
    return decision


@app.post("/tickets/{ticket_id}/approve")
def approve(ticket_id: str):
    with sqlite3.connect(QUEUE_PATH) as db:
        changed = db.execute("UPDATE queue SET status = 'approved' WHERE id = ? AND status = 'pending'",
                             (ticket_id,)).rowcount
    if not changed:
        raise HTTPException(404, "no pending decision for this ticket")
    return {"id": ticket_id, "status": "approved"}      # only now does your help desk send the reply
```
