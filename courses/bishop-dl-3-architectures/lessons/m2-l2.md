# Build: convolutions in minigrad, and a CNN for digits

## 1. Add convolution to your library

The trick that makes convolution fast in NumPy is called **im2col**: cut every k × k patch out of the image into a row of
a big matrix, then do **one** matrix product with the filters. The backward pass does the same in reverse: the filter
gradient is a matrix product, and each patch's input gradient is added back to the pixels it was cut from.

Add this to the end of your `minigrad.py` (it's also in this course's files):

<!-- skip -->
```python
def conv2d(x: Tensor, W: Tensor, b: Tensor = None, padding=0) -> Tensor:
    """x: (N, C, H, W) images, W: (F, C, k, k) filters, b: (F,). Stride 1. Returns (N, F, H', W')."""
    xp = np.pad(x.data, ((0, 0), (0, 0), (padding, padding), (padding, padding)))
    F, C, k, _ = W.shape
    patches = np.lib.stride_tricks.sliding_window_view(xp, (k, k), axis=(2, 3))   # (N, C, H', W', k, k)
    N, _, Ho, Wo = patches.shape[:4]
    cols = patches.transpose(0, 2, 3, 1, 4, 5).reshape(N * Ho * Wo, C * k * k)      # one row per output position
    out = cols @ W.data.reshape(F, -1).T                                            # (N*H'*W', F)
    if b is not None:
        out = out + b.data
    out = out.reshape(N, Ho, Wo, F).transpose(0, 3, 1, 2)

    def back(g):                                                                    # g: (N, F, H', W')
        g2 = g.transpose(0, 2, 3, 1).reshape(-1, F)
        W._add_grad((g2.T @ cols).reshape(W.shape))
        if b is not None:
            b._add_grad(g2.sum(0))
        if x.requires_grad:
            dcols = (g2 @ W.data.reshape(F, -1)).reshape(N, Ho, Wo, C, k, k)
            dxp = np.zeros_like(xp)
            for i in range(k):                                                     # send each patch's gradient back
                for j in range(k):                                                 # to the pixels it was cut from
                    dxp[:, :, i:i + Ho, j:j + Wo] += dcols[:, :, :, :, i, j].transpose(0, 3, 1, 2)
            x._add_grad(dxp[:, :, padding:padding + x.shape[2], padding:padding + x.shape[3]])
    parents = (x, W) if b is None else (x, W, b)
    return x._make(out, parents, back)


class Conv2d(Module):
    def __init__(self, c_in, c_out, k, rng, padding=0):
        self.W = Tensor(rng.normal(0, np.sqrt(2.0 / (c_in * k * k)), (c_out, c_in, k, k)), requires_grad=True)
        self.b = Tensor(np.zeros(c_out), requires_grad=True)
        self.padding = padding

    def forward(self, x):
        return conv2d(x, self.W, self.b, self.padding)


def max_pool2(x: Tensor) -> Tensor:
    """2 × 2 max pooling with stride 2 on (N, C, H, W), H and W even."""
    N, C, H, W = x.shape
    return x.reshape(N, C, H // 2, 2, W // 2, 2).max(axis=5, keepdims=False).max(axis=3, keepdims=False)
```

Pooling needs no new backward code: it's a reshape followed by two `max` reductions, which your library already knows.

## 2. Check the gradients

```python
import numpy as np
from minigrad import Tensor, conv2d, max_pool2

rng = np.random.default_rng(0)
x0, W0, b0 = rng.normal(size=(2, 2, 6, 6)), rng.normal(size=(3, 2, 3, 3)), rng.normal(size=3)

def f(xa, Wa, ba, track=False):
    x, W, b = Tensor(xa, track), Tensor(Wa, track), Tensor(ba, track)
    return (x, W, b), (max_pool2(conv2d(x, W, b, padding=1).tanh()) ** 2).sum()

(x, W, b), loss = f(x0, W0, b0, track=True)
loss.backward()

def numeric(which):
    arrays = [x0, W0, b0]
    g = np.zeros_like(arrays[which])
    for i in np.ndindex(g.shape):
        up = [a.copy() for a in arrays]; up[which][i] += 1e-6
        down = [a.copy() for a in arrays]; down[which][i] -= 1e-6
        g[i] = (float(f(*up)[1].data) - float(f(*down)[1].data)) / 2e-6
    return g

err = max(np.abs(t.grad - numeric(k)).max() for k, t in enumerate([x, W, b]))
print(f"largest gradient error: {err:.1e}")
```

```output
largest gradient error: 1.0e-08
```

## 3. A small CNN for 8 × 8 digits

Two convolution blocks (convolution, ReLU, 2 × 2 max pooling) shrink 8 × 8 to 4 × 4 to 2 × 2, then a linear layer
classifies the 16 × 2 × 2 = 64 features.

```python
from sklearn.datasets import load_digits
from minigrad import Module, Conv2d, Linear, MLP, Adam, cross_entropy

digits = load_digits()
X, y = digits.images / 16.0, digits.target
idx = np.random.default_rng(0).permutation(len(y))
tr, va, te = idx[:1078], idx[1078:1437], idx[1437:]

class CNN(Module):
    def __init__(self, rng):
        self.c1 = Conv2d(1, 8, 3, rng, padding=1)
        self.c2 = Conv2d(8, 16, 3, rng, padding=1)
        self.fc = Linear(16 * 2 * 2, 10, rng)

    def forward(self, x):                       # x: (N, 1, 8, 8)
        h = max_pool2(self.c1(x).relu())        # (N, 8, 4, 4)
        h = max_pool2(self.c2(h).relu())        # (N, 16, 2, 2)
        return self.fc(h.reshape(h.shape[0], -1))

def as_input(images, flat):
    return Tensor(images.reshape(len(images), -1)) if flat else Tensor(images[:, None])

def accuracy(model, images, labels, flat=False):
    return np.mean(model(as_input(images, flat)).data.argmax(axis=1) == labels)

def random_shift(images, r):
    """Move each image by -1, 0 or +1 pixels in each direction (data augmentation, section 9.1.3)."""
    out = images.copy()
    for i in range(len(out)):
        out[i] = np.roll(out[i], (r.integers(-1, 2), r.integers(-1, 2)), axis=(0, 1))
    return out

def train(model, flat=False, augment=False, epochs=30):
    opt, r = Adam(model.parameters(), lr=3e-3), np.random.default_rng(1)
    best, best_w = -1.0, None
    for epoch in range(epochs):
        order = r.permutation(tr)
        for s in range(0, len(order), 64):
            rows = order[s:s + 64]
            batch = random_shift(X[rows], r) if augment else X[rows]
            loss = cross_entropy(model(as_input(batch, flat)), y[rows])
            opt.zero_grad(); loss.backward(); opt.step()
        val = accuracy(model, X[va], y[va], flat)
        if val > best:                                   # early stopping
            best, best_w = val, [p.data.copy() for p in model.parameters()]
    for p, w in zip(model.parameters(), best_w):
        p.data = w
    return model
```

## 4. Fully connected versus convolutional, with and without a shift

Train three models, then test each on the normal test images **and** on the same images moved one pixel to the right.

```python
shifted_te = np.roll(X[te], 1, axis=2)
models = {
    "fully connected (part 2)": (train(MLP([64, 128, 10], np.random.default_rng(0)), flat=True), True),
    "CNN": (train(CNN(np.random.default_rng(0))), False),
    "CNN + shift augmentation": (train(CNN(np.random.default_rng(0)), augment=True, epochs=60), False),
}
print(f"{'model':26s} {'test':>6s} {'shifted test':>13s}")
for name, (model, flat) in models.items():
    print(f"{name:26s} {accuracy(model, X[te], y[te], flat):6.3f} {accuracy(model, shifted_te, y[te], flat):13.3f}")
```

```output
model                        test  shifted test
fully connected (part 2)    0.983         0.489
CNN                         0.981         0.606
```

## Read the table

- On normal images, all three do well.
- On shifted images, the fully connected network collapses: for it, a shifted digit is a different set of 64 numbers.
- The plain CNN keeps more of its accuracy, thanks to equivariant convolutions and pooling, but its final linear layer
  still cares where the features land.
- Training on randomly shifted copies (augmentation, section 9.1.3) teaches the CNN to ignore the shift, at a small cost
  on the original images.

That's the chapter 9 to chapter 10 story in one table: build in what you can (convolutions), and teach the rest with data.
