# Project part 1: a decoder transformer on minigrad

## 1. The data

```python
import numpy as np
from minigrad import Tensor, Module, Linear, Adam, softmax, cross_entropy

L = 6                     # digits to sort
SEP = 10                  # the separator token
V = 11                    # vocabulary: digits 0-9 and the separator
T = 2 * L                 # the model sees 12 positions and predicts the next token at each

def make_batch(rng, B):
    digits = rng.integers(0, 10, (B, L))
    seq = np.concatenate([digits, np.full((B, 1), SEP), np.sort(digits, axis=1)], axis=1)   # 13 tokens
    return seq[:, :-1], seq[:, 1:]          # inputs, and the next token at every position

inp, tgt = make_batch(np.random.default_rng(0), 2)
print("inputs: \n", inp)
print("targets:\n", tgt)
```

```output
inputs: 
 [[ 8  6  5  2  3  0 10  0  2  3  5  6]
 [ 0  0  1  8  6  9 10  0  0  1  6  8]]
targets:
 [[ 6  5  2  3  0 10  0  2  3  5  6  8]
 [ 0  1  8  6  9 10  0  0  1  6  8  9]]
```

The targets are the inputs moved one step to the left: at every position, the next token.

## 2. Layer normalization (section 7.4.3)

Normalize each token's vector to mean 0 and variance 1, then let the network re-scale it with learned `g` and `b`.

```python
class LayerNorm(Module):
    def __init__(self, d):
        self.g = Tensor(np.ones(d), requires_grad=True)
        self.b = Tensor(np.zeros(d), requires_grad=True)

    def forward(self, x):
        centred = x - x.mean(axis=-1, keepdims=True)
        var = (centred * centred).mean(axis=-1, keepdims=True)
        return centred * (var + 1e-5) ** -0.5 * self.g + self.b
```

## 3. Multi-head causal self-attention (sections 12.1.6 and 12.3.1)

The same reshape-and-transpose trick as on day 3, now with Tensors, so every step is differentiable. One matrix computes
queries, keys and values together; the mask hides future positions.

```python
class CausalSelfAttention(Module):
    def __init__(self, d, heads, rng):
        self.heads = heads
        self.qkv = Linear(d, 3 * d, rng, bias=False)
        self.out = Linear(d, d, rng)
        self.last_attention = None                       # kept so we can look inside later

    def forward(self, x):
        B, N, d = x.shape
        dh = d // self.heads
        qkv = self.qkv(x)
        def split(t):                                    # (B, N, d) -> (B, heads, N, dh)
            return t.reshape(B, N, self.heads, dh).transpose(0, 2, 1, 3)
        q, k, v = split(qkv[:, :, :d]), split(qkv[:, :, d:2 * d]), split(qkv[:, :, 2 * d:])
        scores = q @ k.transpose(0, 1, 3, 2) * (1 / np.sqrt(dh))
        mask = np.triu(np.full((N, N), -1e9), k=1)       # no looking ahead
        att = softmax(scores + mask, axis=-1)
        self.last_attention = att.data
        y = (att @ v).transpose(0, 2, 1, 3).reshape(B, N, d)
        return self.out(y)
```

## 4. The transformer layer (section 12.1.7, Figure 12.9)

Layer norm, attention, residual; layer norm, MLP, residual. This is the "pre-norm" arrangement most modern models use.

```python
class TransformerLayer(Module):
    def __init__(self, d, heads, rng):
        self.norm1, self.attn = LayerNorm(d), CausalSelfAttention(d, heads, rng)
        self.norm2, self.fc1, self.fc2 = LayerNorm(d), Linear(d, 4 * d, rng), Linear(4 * d, d, rng)

    def forward(self, x):
        x = x + self.attn(self.norm1(x))
        return x + self.fc2(self.fc1(self.norm2(x)).relu())
```

## 5. The whole model

Token embeddings plus learned position embeddings, a stack of layers, a final layer norm, and a linear layer that gives
a score for every token in the vocabulary (Figure 12.15).

```python
class TinyGPT(Module):
    def __init__(self, rng, d=48, heads=4, n_layers=2):
        self.tok = Tensor(rng.normal(0, 0.1, (V, d)), requires_grad=True)
        self.pos = Tensor(rng.normal(0, 0.1, (T, d)), requires_grad=True)
        self.layers = [TransformerLayer(d, heads, rng) for _ in range(n_layers)]
        self.norm = LayerNorm(d)
        self.head = Linear(d, V, rng)

    def forward(self, idx):                     # idx: (B, N) integer tokens
        x = self.tok[idx] + self.pos[:idx.shape[1]]
        for layer in self.layers:
            x = layer(x)
        return self.head(self.norm(x))          # (B, N, V) scores for the next token

model = TinyGPT(np.random.default_rng(1))
print("parameters:", sum(p.data.size for p in model.parameters()))
print("output shape for the batch above:", model(inp).shape)
```

```output
parameters: 57995
output shape for the batch above: (2, 12, 11)
```

`self.tok[idx]` is the embedding lookup: indexing a Tensor with an array of token ids picks one row per token, and your
library's `__getitem__` sends the gradients back to the right rows.

## 6. Check the gradients of the whole model

Before training, one gradient check on a few parameters of the complete transformer. If attention or layer norm had a
bug, it would show here.

```python
def loss_fn(m, inp, tgt):
    logits = m(inp)[:, L:, :]                            # score only the sorted half
    return cross_entropy(logits.reshape(-1, V), tgt[:, L:].reshape(-1))

loss = loss_fn(model, inp, tgt)
for p in model.parameters():
    p.grad = None
loss.backward()

worst = 0.0
for p in [model.tok, model.layers[0].attn.qkv.W, model.layers[1].norm2.g]:
    for i in list(np.ndindex(p.shape))[:5]:
        old = p.data[i]
        p.data[i] = old + 1e-6; up = float(loss_fn(model, inp, tgt).data)
        p.data[i] = old - 1e-6; down = float(loss_fn(model, inp, tgt).data)
        p.data[i] = old
        worst = max(worst, abs((up - down) / 2e-6 - p.grad[i]))
print(f"largest gradient error: {worst:.1e}")
```

```output
largest gradient error: 3.5e-10
```

## 7. Train it

Fresh random lists every step, so the model can never memorize: it has to learn to sort.

```python
rng = np.random.default_rng(0)
opt = Adam(model.parameters(), lr=3e-3)
for step in range(1, 1501):
    inp, tgt = make_batch(rng, 64)
    loss = loss_fn(model, inp, tgt)
    opt.zero_grad()
    loss.backward()
    opt.step()
    if step % 250 == 0:
        print(f"step {step:4d}: loss {float(loss.data):.4f}")
```

```output

step  500: loss 0.0159
step  750: loss 0.0015
```

## 8. Test it: greedy decoding

Give it new lists, then generate the sorted half one token at a time, always taking the most probable token (greedy
search, section 12.3.2). Count a list as right only if all six outputs are right.

```python
def sort_with_model(m, digits):
    seq = list(digits) + [SEP]
    for _ in range(L):
        logits = m(np.array([seq]))
        seq.append(int(logits.data[0, -1].argmax()))
    return seq[L + 1:]

tests = np.random.default_rng(99).integers(0, 10, (300, L))
correct = sum(sort_with_model(model, d) == sorted(d.tolist()) for d in tests)
print(f"lists sorted perfectly: {correct} of {len(tests)}")
for d in tests[:3]:
    print(d, "->", sort_with_model(model, d))
```

```output
lists sorted perfectly: 299 of 300
[9 5 7 5 1 5] -> [1, 5, 5, 5, 7, 9]
[9 9 9 6 3 5] -> [3, 5, 6, 9, 9, 9]
[6 2 9 5 3 4] -> [2, 3, 4, 5, 6, 9]
```

## 9. Look inside: where does it attend?

When the model writes the k-th sorted digit, which input positions does it look at? Print, for one example, the input
position that gets the most attention in the last layer (averaged over its heads) at each output step.

```python
d = np.array([7, 2, 9, 2, 0, 5])
seq = np.array([list(d) + [SEP] + sorted(d.tolist())[:-1]])
model(seq)
att = model.layers[-1].attn.last_attention[0].mean(axis=0)       # (12, 12), averaged over heads
print("input digits:", d)
for k in range(L):
    row = att[L + k, :L]                                          # the step that predicts output k, looking at the inputs
    print(f"writing output {k + 1} ({sorted(d.tolist())[k]}): strongest attention on input position "
          f"{row.argmax() + 1} (digit {d[row.argmax()]}), weight {row.max():.2f}")
```

```output
input digits: [7 2 9 2 0 5]
writing output 1 (0): strongest attention on input position 6 (digit 5), weight 0.23
writing output 2 (2): strongest attention on input position 4 (digit 2), weight 0.30
writing output 3 (2): strongest attention on input position 2 (digit 2), weight 0.23
writing output 4 (5): strongest attention on input position 6 (digit 5), weight 0.74
writing output 5 (7): strongest attention on input position 1 (digit 7), weight 0.41
writing output 6 (9): strongest attention on input position 3 (digit 9), weight 0.33
```

Read it slowly. In five of the six steps, the strongest attention lands on an input holding exactly the digit about to be
written: the model has learned to **find** the next smallest number with attention. (The first step is the exception:
averaging over heads hides that the information can also come through the first layer.) Nobody programmed a sorting
algorithm; it emerged from predicting the next token.
