# A test database and fake users

Tests should never touch your real database. Override `get_db` with a fresh database for each test:

```python
# conftest.py
import pytest
from fastapi.testclient import TestClient
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from sqlalchemy.pool import StaticPool

from app.main import app
from app.db import Base, get_db
from app.auth import get_current_user
from app.models import User


@pytest.fixture
def db():
    engine = create_engine(
        "sqlite://",                                  # in memory
        connect_args={"check_same_thread": False},
        poolclass=StaticPool,                         # one shared connection, so tables stay
    )
    Base.metadata.create_all(engine)
    Session = sessionmaker(engine)
    with Session() as session:
        yield session


@pytest.fixture
def client(db):
    app.dependency_overrides[get_db] = lambda: db
    with TestClient(app) as c:
        yield c
    app.dependency_overrides.clear()


@pytest.fixture
def as_user(client, db):
    user = User(email="a@test.com", password_hash="x", role="editor")
    db.add(user); db.commit()
    app.dependency_overrides[get_current_user] = lambda: user
    return user
```

```python
def test_editor_can_create(client, as_user):
    r = client.post("/books", json={"title": "Emma", "author_id": 1})
    assert r.status_code == 201
```

Each test gets a clean database, and you choose who is "logged in" without creating real tokens.

> SQLite is fast for tests, but if production uses PostgreSQL, also run your tests against PostgreSQL (for example
> in CI with a service container). Some SQL behaves differently.
