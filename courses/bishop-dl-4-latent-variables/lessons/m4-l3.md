# Build: K-means from scratch, and an image with 8 colours

## 1. K-means

```python
import numpy as np

def kmeans(X, K, rng, iters=100):
    centres = X[rng.choice(len(X), K, replace=False)].copy()   # start at K random data points
    history = []
    for _ in range(iters):
        d = ((X[:, None, :] - centres[None, :, :]) ** 2).sum(-1)
        assign = d.argmin(axis=1)                               # step 1: nearest centre
        history.append(d[np.arange(len(X)), assign].sum())      # the distortion J
        new = np.array([X[assign == k].mean(0) if np.any(assign == k) else centres[k] for k in range(K)])
        if np.allclose(new, centres):
            break
        centres = new                                           # step 2: move centres to their means
    return centres, assign, history

rng = np.random.default_rng(0)
blobs = np.vstack([rng.normal(c, 0.6, (100, 2)) for c in ([0, 0], [4, 0], [2, 3])])
centres, assign, J = kmeans(blobs, 3, rng)
print("centres:\n", np.round(centres, 2))
print("distortion at each iteration:", np.round(J, 1))
```

```output
centres:
 [[ 3.93 -0.05]
 [ 1.98  3.03]
 [-0.04  0.06]]
distortion at each iteration: [1536.9  761.2  244.   213.2]
```

The distortion only ever goes down, and the centres land near the three true blob centres (0, 0), (4, 0) and (2, 3).

## 2. Local minima

Run it from many random starts on a harder problem and look at the final distortion of each.

```python
blobs6 = np.vstack([rng.normal(c, 0.5, (60, 2)) for c in ([0, 0], [3, 0], [6, 0], [0, 3], [3, 3], [6, 3])])
finals = sorted(round(float(kmeans(blobs6, 6, np.random.default_rng(s))[2][-1]), 1) for s in range(10))
print("final distortion from 10 random starts:", finals)
```

```output
final distortion from 10 random starts: [167.0, 167.0, 167.0, 167.0, 167.0, 167.0, 167.0, 167.0, 396.3, 403.9]
```

Two of the ten runs get stuck with a much worse distortion: two centres share one blob while another centre covers two. The fix is
simple: run several starts and keep the best.

## 3. Compress a photo to 8 colours (section 15.1.1)

scikit-learn ships a sample photo. Treat every pixel as a point in colour space (red, green, blue) and cluster.

```python
from sklearn.datasets import load_sample_image

img = load_sample_image("china.jpg") / 255.0               # 427 × 640 pixels, 3 colours each
pixels = img.reshape(-1, 3)
sample = pixels[rng.choice(len(pixels), 5000, replace=False)]   # fit on a sample, for speed
centres, _, _ = kmeans(sample, 8, rng)

d = ((pixels[:, None, :] - centres[None, :, :]) ** 2).sum(-1)
labels = d.argmin(axis=1)
compressed = centres[labels].reshape(img.shape)

error = np.sqrt(np.mean((compressed - img) ** 2))
bits_original = pixels.size * 8
bits_compressed = len(pixels) * 3 + 8 * 3 * 8              # 3 bits per pixel index, plus the 8 colours
print(f"root-mean-square colour error: {error:.3f} (on a 0 to 1 scale)")
print(f"size: {bits_original:,} bits -> {bits_compressed:,} bits  ({bits_original / bits_compressed:.1f} times smaller)")
```

```output
root-mean-square colour error: 0.059 (on a 0 to 1 scale)
size: 6,558,720 bits -> 820,032 bits  (8.0 times smaller)
```

Eight times smaller, and the picture is still clearly recognizable (try `plt.imshow(compressed)`). That's the trade-off of
lossy compression in section 15.1.1: fewer colours, smaller file, more error.

## 4. Soft assignments

K-means gives each point to exactly one cluster. A Gaussian mixture says "70% this one, 30% that one" for points between
clusters. Compute the responsibilities for a point near halfway between two of the first blobs, using equal-weight, equal-width
Gaussians on the K-means centres.

```python
centres, _, _ = kmeans(blobs, 3, np.random.default_rng(0))
left = centres[np.argmin(((centres - [0, 0]) ** 2).sum(1))]     # the centre near (0, 0)
right = centres[np.argmin(((centres - [4, 0]) ** 2).sum(1))]    # the centre near (4, 0)
point = (left + right) / 2 + np.array([0.1, 0.0])               # just right of halfway
d2 = ((point - centres) ** 2).sum(-1)
resp = np.exp(-d2 / (2 * 0.6 ** 2))
resp /= resp.sum()
print("point:", np.round(point, 2))
for c, r in zip(centres, resp):
    print(f"  centre {np.round(c, 2)}: responsibility {r:.2f}")
```

```output
point: [2.04 0.01]
  centre [-0.04  0.06]: responsibility 0.25
  centre [ 3.92 -0.03]: responsibility 0.75
  centre [1.96 3.04]: responsibility 0.00
```
