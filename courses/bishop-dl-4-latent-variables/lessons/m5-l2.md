# Project part 1: EM for a Gaussian mixture, from scratch

## 1. Data the clusters don't make easy

Three groups: two are long, tilted ellipses, and one is a small round blob. K-means, which only knows round clusters,
will struggle.

```python
import numpy as np

    rng.multivariate_normal([2.5, 4], [[0.2, 0], [0, 0.2]], 100),
])
truth = np.repeat([0, 1, 2], [300, 300, 100])
print(X.shape)
```

## 2. The pieces of EM

The Gaussian density in log form (safer than exp-then-log), then the E step and the M step from section 15.3.1.

```python
def log_gauss(X, mu, S):
    D = X.shape[1]
    L = np.linalg.cholesky(S)
    sol = np.linalg.solve(L, (X - mu).T)
    return -0.5 * (D * np.log(2 * np.pi) + 2 * np.log(np.diag(L)).sum() + (sol ** 2).sum(0))

def e_step(X, pi, mu, S):
    logp = np.stack([np.log(pi[k]) + log_gauss(X, mu[k], S[k]) for k in range(len(pi))], axis=1)
    top = logp.max(axis=1, keepdims=True)
    log_px = top[:, 0] + np.log(np.exp(logp - top).sum(axis=1))   # log p(xₙ), by log-sum-exp
    return np.exp(logp - log_px[:, None]), log_px.sum()           # responsibilities, log-likelihood

def m_step(X, gamma, floor=1e-6):
    Nk = gamma.sum(axis=0)
    pi = Nk / len(X)
    mu = (gamma.T @ X) / Nk[:, None]
    S = []
    for k in range(gamma.shape[1]):
        d = X - mu[k]
        S.append((gamma[:, k, None] * d).T @ d / Nk[k] + floor * np.eye(X.shape[1]))   # small floor: no collapse
    return pi, mu, np.array(S)
```

The tiny `floor` added to each covariance is the guard against the singularities of section 15.2: a component can never
shrink to a single point.

## 3. Run it, and watch the log-likelihood

```python
def em(X, K, rng, iters=200, tol=1e-6):
    mu = X[rng.choice(len(X), K, replace=False)]
    S = np.array([np.cov(X.T) for _ in range(K)])
    pi = np.full(K, 1 / K)
    history = []
    for _ in range(iters):
        gamma, ll = e_step(X, pi, mu, S)
        history.append(ll)
        if len(history) > 1 and history[-1] - history[-2] < tol:
            break
        pi, mu, S = m_step(X, gamma)
    return pi, mu, S, gamma, history

pi, mu, S, gamma, history = em(X, 3, np.random.default_rng(0))
steps = np.diff(history)
print(f"iterations: {len(history)}   final log-likelihood: {history[-1]:.2f}")
print(f"did the log-likelihood ever go down? {bool(np.any(steps < -1e-9))}")
print("mixing weights:", np.round(np.sort(pi), 3), "  (truth: 0.143, 0.429, 0.429)")
```

```output
iterations: 93   final log-likelihood: -2718.50
did the log-likelihood ever go down? False
mixing weights: [0.139 0.43  0.431]   (truth: 0.143, 0.429, 0.429)
```

The log-likelihood never decreased: section 15.4's guarantee, checked on real numbers.

## 4. Compare with K-means

Measure how well each method recovers the true groups. Because cluster numbers are arbitrary, match each found cluster to
the true group it overlaps most.

```python
def agreement(labels):
    return sum(np.bincount(truth[labels == c], minlength=3).max() for c in np.unique(labels)) / len(truth)

def kmeans_labels(X, K, rng, iters=100):
    c = X[rng.choice(len(X), K, replace=False)]
    for _ in range(iters):
        lab = ((X[:, None] - c[None]) ** 2).sum(-1).argmin(1)
        c = np.array([X[lab == k].mean(0) for k in range(K)])
    return lab

print(f"K-means agreement with the true groups: {agreement(kmeans_labels(X, 3, np.random.default_rng(0))):.3f}")
print(f"EM (Gaussian mixture) agreement:        {agreement(gamma.argmax(axis=1)):.3f}")
```

```output
K-means agreement with the true groups: 0.753
EM (Gaussian mixture) agreement:        0.926
```

K-means cuts the tilted ellipses with straight, round boundaries. The mixture learns each group's own covariance, so its
boundaries follow the shapes.
