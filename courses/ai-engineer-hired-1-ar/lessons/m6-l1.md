# الواجهة البرمجية والاختبارات وDocker وCI

غلّف النظام بواجهة برمجية صغيرة ليستخدمه موقع أو روبوت أو خدمة أخرى. ويُحمَّل الفهرس مرة واحدة عند بدء الخادم:

```python
# docanswer/api.py
from contextlib import asynccontextmanager

from fastapi import FastAPI
from pydantic import BaseModel

from docanswer.answer import answer
from docanswer.search import Searcher

state = {}


@asynccontextmanager
async def lifespan(app):
    state["searcher"] = Searcher()          # load the index once, when the server starts
    yield


app = FastAPI(title="DocAnswer", lifespan=lifespan)


class Ask(BaseModel):
    question: str


@app.post("/ask")
def ask(body: Ask):
    result = answer(body.question, state["searcher"])
    return {"answer": result["answer"],
            "sources": [{"source": c["source"], "page": c["page"]} for c in result["citations"]]}


@app.get("/health")
def health():
    return {"ok": True}
```

## اختبارات لا تحتاج إلى نموذج

يجب أن تكون اختبارات الوحدة سريعة ومجانية. استبدل نموذج التضمين والنموذج اللغوي بنسخ وهمية صغيرة، واختبر منطقك أنت: التقطيع، والدمج، وتصفية الإشارات، وعقد الواجهة البرمجية. (أما جودة النموذج فتقيسها تقييماتك.)

```python
# tests/test_core.py
import json

import numpy as np
from fastapi.testclient import TestClient

from docanswer import answer as answer_module
from docanswer.ingest import chunk
from docanswer.search import Searcher, rrf


def test_chunk_packs_paragraphs_with_overlap():
    text = "\n\n".join(f"Paragraph {i} " + "x" * 300 for i in range(6))
    parts = chunk(text, size=800, overlap=150)
    assert len(parts) > 1
    assert all(len(p) <= 800 + 150 + 320 for p in parts)
    assert parts[1].startswith(parts[0][-150:].strip()[:20])


def test_rrf_rewards_items_ranked_well_in_both_lists():
    assert rrf([[1, 2, 3], [3, 1, 4]], k=2) == [1, 3]


class FakeModel:                       # turns text into a bag-of-letters vector: enough to test the plumbing
    def encode(self, texts, normalize_embeddings=True):
        v = np.array([[t.lower().count(ch) for ch in "abcdefghijklmnopqrstuvwxyz"] for t in texts], dtype=float) + 1e-9
        return v / np.linalg.norm(v, axis=1, keepdims=True)


def make_searcher(tmp_path):
    rows = [{"id": f"doc:{i}:0", "source": "doc", "page": i, "text": t} for i, t in enumerate(
        ["Annual leave is 21 days per year.", "Sick leave needs a doctor's note.", "The office opens at 9 am."])]
    path = tmp_path / "chunks.jsonl"
    path.write_text("\n".join(json.dumps(r) for r in rows), encoding="utf-8")
    return Searcher(str(path), model=FakeModel())


def test_search_finds_the_right_chunk(tmp_path):
    s = make_searcher(tmp_path)
    assert s.search("how many days of annual leave", k=1)[0]["page"] == 0


class FakeClient:
    class chat:
        class completions:
            @staticmethod
            def create(**kwargs):
                class R:
                    class choices0:
                        class message:
                            content = "You get 21 days [1]."
                    choices = [choices0]
                    usage = None
                return R


def test_answer_keeps_only_real_citations(tmp_path, monkeypatch):
    monkeypatch.setattr(answer_module, "client", FakeClient)
    result = answer_module.answer("annual leave days?", make_searcher(tmp_path), k=2)
    assert result["answer"].endswith("[1].")
    assert len(result["citations"]) == 1


def test_api(tmp_path, monkeypatch):
    from docanswer import api
    monkeypatch.setattr(api, "Searcher", lambda: make_searcher(tmp_path))
    monkeypatch.setattr(api, "answer", lambda q, s: {"answer": "21 days [1]", "citations": [{"source": "doc", "page": 0}]})
    with TestClient(api.app) as client:
        r = client.post("/ask", json={"question": "leave?"})
        assert r.status_code == 200 and r.json()["sources"] == [{"source": "doc", "page": 0}]
```

# Dockerfile
FROM python:3.12-slim

EXPOSE 8000
CMD ["uvicorn", "docanswer.api:app", "--host", "0.0.0.0", "--port", "8000"]
