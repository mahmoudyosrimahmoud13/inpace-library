# Pure ASGI middleware: inPace's NavMiddleware

Middleware wraps the app. inPace's is written as pure ASGI (not `BaseHTTPMiddleware`), so it can change the response
on its way out without buffering the body:

```python
class NavMiddleware:
    def __init__(self, app):
        self.app = app

    async def __call__(self, scope, receive, send):
        if scope["type"] != "http":
            return await self.app(scope, receive, send)
        nav = any(k == b"x-inpace-nav" for k, _ in scope.get("headers", []))
        state = {"swallow": False}

        async def wrapped(msg):
            if msg["type"] == "http.response.start":
                headers = list(msg.get("headers", []))
                found = {k.lower(): v for k, v in headers}
                if nav and msg["status"] in (301, 302, 303, 307, 308) and b"location" in found:
                    body = json.dumps({"go": found[b"location"].decode("latin-1")}).encode()
                    state["swallow"] = True            # drop the original redirect's body
                    await send({"type": "http.response.start", "status": 200, "headers": [
                        (b"content-type", b"application/json"), (b"cache-control", b"no-store"),
                        (b"content-length", str(len(body)).encode())]})
                    await send({"type": "http.response.body", "body": body})
                    return
                if found.get(b"content-type", b"").startswith(b"text/html") and b"cache-control" not in found:
                    headers.append((b"cache-control", b"no-store"))
                    msg = {**msg, "headers": headers}
                await send(msg)
            elif not state["swallow"]:
                await send(msg)

        await self.app(scope, receive, wrapped)
```

The trick: **wrap `send`**. The route still returns a normal redirect, and the middleware rewrites it into JSON for
the page script. Routes stay simple; the rule lives in one place. The real one also reads the `tz` cookie to set the
clock.
