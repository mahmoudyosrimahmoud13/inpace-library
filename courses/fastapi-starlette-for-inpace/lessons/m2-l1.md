# Routes and the app factory

```python
from starlette.applications import Starlette
from starlette.requests import Request
from starlette.responses import HTMLResponse, JSONResponse, RedirectResponse
from starlette.routing import Mount, Route
from starlette.staticfiles import StaticFiles

def create_app(vault) -> Starlette:
    V = vault

    def go(url, ok=None, err=None):
        ...  # builds "/path?ok=...#section" and returns RedirectResponse(url, 303)

    async def lesson_done(request: Request):
        jid, lid = request.path_params["jid"], request.path_params["lid"]
        ...
        form = await request.form()
        target = form.get("next")
        return go(f"/learn/{jid}/lesson/{target}", yay="lesson")

    def needs_setup(handler):
        async def wrapped(request):
            if not V.is_set_up:
                return go("/setup")
            return await handler(request)
        return wrapped

    routes = [
        Route("/learn/{jid}/lesson/{lid}/done", needs_setup(lesson_done), methods=["POST"]),
        Mount("/static", StaticFiles(directory=HERE / "static")),
    ]
    return Starlette(routes=routes, middleware=[Middleware(NavMiddleware)])
```

## Why an app factory

`create_app(vault)` takes its dependencies as arguments. Tests create a fresh vault in a temp folder and a fresh
app for each test, with no global state to reset. The closure (`V`) acts as simple dependency injection.

## Path parameters

`{jid}` is a string; `{n:int}` converts; `{path:path}` matches slashes (inPace uses it for course media).
