Then the **logic**, in its own file with no DOM, so you can test it later with Node. Each function returns a new array instead of changing the old one:

```javascript
// tasks.js
// Pure logic, no DOM: easy to test with node.
export function createTask(title, minutes = 25) {
  const clean = title.trim();
  if (!clean) throw new Error("A task needs a title.");
  return { id: Date.now().toString(36) + Math.random().toString(36).slice(2, 6), title: clean, minutes: Number(minutes), done: false };
}

export function toggleTask(tasks, id) {
  return tasks.map((t) => (t.id === id ? { ...t, done: !t.done } : t));
}

export function removeTask(tasks, id) {
  return tasks.filter((t) => t.id !== id);
}

export function summary(tasks) {
  const left = tasks.filter((t) => !t.done);
  const minutes = left.reduce((sum, t) => sum + t.minutes, 0);
  return { total: tasks.length, left: left.length, minutes };
}
```
