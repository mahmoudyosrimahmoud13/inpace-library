# Project step 2: connect, save, load a quote

**localStorage** keeps text in the browser between visits. Save the tasks as JSON:

```javascript
// storage.js
const KEY = "study-tasks";

export function load() {
  try {
    return JSON.parse(localStorage.getItem(KEY)) ?? [];
  } catch {
    return [];
  }
}

export function save(tasks) {
  localStorage.setItem(KEY, JSON.stringify(tasks));
}
```

Now `app.js` connects everything: it draws the list from the data (`render`), handles the form and clicks with delegation, saves after each change, and loads a quote with fetch, with a fallback when there's no internet:

```javascript
// app.js
import { createTask, toggleTask, removeTask, summary } from "./tasks.js";
import { load, save } from "./storage.js";

let tasks = load();
const form = document.querySelector("#task-form");
const input = document.querySelector("#task-input");
const minutes = document.querySelector("#task-minutes");
const list = document.querySelector("#task-list");
const info = document.querySelector("#summary");

function render() {
  list.innerHTML = "";
  for (const task of tasks) {
    const li = document.createElement("li");
    li.className = task.done ? "done" : "";
    li.dataset.id = task.id;
    const box = document.createElement("input");
    box.type = "checkbox";
    box.checked = task.done;
    const text = document.createElement("span");
    text.textContent = `${task.title} (${task.minutes} min)`;   // textContent, never innerHTML, for user text
    const del = document.createElement("button");
    del.textContent = "Delete";
    del.className = "delete";
    li.append(box, text, del);
    list.append(li);
  }
  const s = summary(tasks);
  info.textContent = `${s.left} of ${s.total} left, about ${s.minutes} minutes of study.`;
  save(tasks);
}

form.addEventListener("submit", (event) => {
  event.preventDefault();
  tasks = [...tasks, createTask(input.value, minutes.value)];
  form.reset();
  input.focus();
  render();
});

// One listener for the whole list (event delegation).
list.addEventListener("click", (event) => {
  const id = event.target.closest("li")?.dataset.id;
  if (!id) return;
  if (event.target.matches("input[type=checkbox]")) tasks = toggleTask(tasks, id);
  if (event.target.matches(".delete")) tasks = removeTask(tasks, id);
  render();
});

async function showQuote() {
  const box = document.querySelector("#quote");
  try {
    const res = await fetch("https://dummyjson.com/quotes/random");
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const { quote, author } = await res.json();
    box.textContent = `"${quote}" (${author})`;
  } catch (error) {
    box.textContent = "Keep going: one task at a time.";
    console.warn("No quote today:", error.message);
  }
}

render();
showQuote();
```

Notice the pattern: **data → render**. You never edit the list by hand; you change `tasks` and call `render()`. Frameworks like React are built on the same idea.
