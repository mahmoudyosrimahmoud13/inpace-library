// 3. A tiny event emitter
class Emitter {
  #h = new Map();
  on(type, fn) { (this.#h.get(type) ?? this.#h.set(type, new Set()).get(type)).add(fn); return () => this.off(type, fn); }
  off(type, fn) { this.#h.get(type)?.delete(fn); }
  emit(type, ...args) { this.#h.get(type)?.forEach((fn) => fn(...args)); }
}
// 4. Promise.all written yourself
function all(list) {
  return new Promise((resolve, reject) => {
    const out = []; let left = list.length;
    if (!left) return resolve(out);
    list.forEach((p, i) => Promise.resolve(p).then((v) => { out[i] = v; if (--left === 0) resolve(out); }, reject));
  });
}
// 5. Retry a fetch up to 3 times with a growing delay
async function retry(url, tries = 3) {
  for (let i = 0; i < tries; i++) {
    try { const r = await fetch(url); if (r.ok) return r; } catch {}
    await new Promise((res) => setTimeout(res, 2 ** i * 300));
  }
  throw new Error("gave up");
}
