# Five implementations to know cold

These come up again and again in ML and AI coding rounds (the [field guide](https://github.com/alexeygrigorev/ai-engineering-field-guide) reports cosine similarity at Amazon and logistic regression with SGD, L2 and early stopping at Mistral). Type each from memory until you can do it in under 15 minutes.

```python
import numpy as np

def top_k_cosine(query, docs, k=3):
    """query: (d,), docs: (n, d). Indexes of the k most similar docs, best first."""
    q = query / np.linalg.norm(query)
    D = docs / np.linalg.norm(docs, axis=1, keepdims=True)
    scores = D @ q
    top = np.argpartition(-scores, k - 1)[:k]      # O(n), not sorted yet
    return top[np.argsort(-scores[top])]


def softmax(z):
    z = z - z.max(axis=-1, keepdims=True)          # no overflow in exp
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)


def cross_entropy(logits, y):
    """logits: (n, c), y: (n,) class indexes."""
    z = logits - logits.max(axis=1, keepdims=True)
    log_probs = z - np.log(np.exp(z).sum(axis=1, keepdims=True))
    return -log_probs[np.arange(len(y)), y].mean()
```

```python
def train_logreg(X, y, X_val, y_val, lr=0.1, l2=1e-3, epochs=200, patience=10, batch=32, seed=0):
    rng = np.random.default_rng(seed)
    w, b = np.zeros(X.shape[1]), 0.0
    sigmoid = lambda t: 1 / (1 + np.exp(-np.clip(t, -30, 30)))

    def loss(Xs, ys):
        p = np.clip(sigmoid(Xs @ w + b), 1e-12, 1 - 1e-12)
        return -np.mean(ys * np.log(p) + (1 - ys) * np.log(1 - p)) + l2 * (w @ w) / 2

    best, best_wb, waited = np.inf, (w.copy(), b), 0
    for _ in range(epochs):
        idx = rng.permutation(len(X))
        for start in range(0, len(X), batch):
            j = idx[start:start + batch]
            err = sigmoid(X[j] @ w + b) - y[j]          # d(loss)/d(logit)
            w -= lr * (X[j].T @ err / len(j) + l2 * w)
            b -= lr * err.mean()
        val = loss(X_val, y_val)
        if val < best - 1e-6:
            best, best_wb, waited = val, (w.copy(), b), 0
        else:
            waited += 1
            if waited >= patience:
                break                                  # early stopping
    return best_wb
```

```python
def kmeans(X, k, iters=100, seed=0):
    rng = np.random.default_rng(seed)
    centers = X[rng.choice(len(X), k, replace=False)]
    for _ in range(iters):
        d = ((X[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2)   # (n, k)
        labels = d.argmin(axis=1)
        new = np.array([X[labels == c].mean(axis=0) if np.any(labels == c) else centers[c]
                        for c in range(k)])
        if np.allclose(new, centers):
            break
        centers = new
    return centers, labels


def attention(Q, K, V, causal=False):
    """Q: (n, d), K: (m, d), V: (m, dv). Scaled dot-product attention."""
    scores = Q @ K.T / np.sqrt(Q.shape[-1])
    if causal:
        n, m = scores.shape
        scores = np.where(np.tril(np.ones((n, m), dtype=bool)), scores, -np.inf)
    return softmax(scores) @ V
```

## What they'll ask next

- *Why divide by √d in attention?* Dot products grow with d; without scaling, softmax saturates and gradients vanish.
- *Why subtract the max in softmax?* `exp` overflows for large inputs; the result is mathematically the same.
- *k-means converged to a bad answer?* It's sensitive to the start: k-means++ or several restarts.
- *Your logistic regression's cost per epoch?* O(n·d).
